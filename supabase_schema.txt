-- ----------------------------------------------------------------------------
-- 1) SURVEY (/survey, /survey/live)
-- ----------------------------------------------------------------------------
create table if not exists survey_responses (
  id uuid primary key default gen_random_uuid(),
  login text not null,
  avatar_url text,
  week_key text not null,
  rating integer not null check (rating between 1 and 5), -- 1-5 bintang kepuasan
  reason text not null,        -- alasan singkat di balik rating bintangnya
  created_at timestamptz not null default now(),
  unique (login, week_key)
);
create index if not exists survey_responses_week_idx
  on survey_responses (week_key, created_at desc);

-- Migrasi aman kalau tabel ini udah ada dari versi survey LAMA (yang masih
-- banyak pertanyaan: opinion/has_issue/issue_detail/suggestion). Survey
-- sekarang disederhanain jadi cuma bintang + alasan. Kolom `opinion`
-- DIPAKAI ULANG jadi `reason` (rename, bukan kolom baru, data lama gak
-- ilang); kolom yang udah gak dipakai (has_issue, issue_detail, suggestion)
-- DIBIARKAN nempel di tabel (histori lama tetep aman) tapi gak wajib diisi
-- lagi. Jalanin blok ini SEKALI aja kalau upgrade dari versi lama:
do $$ begin
  if exists (select 1 from information_schema.columns
             where table_name='survey_responses' and column_name='opinion') then
    alter table survey_responses rename column opinion to reason;
  end if;
end $$;
alter table survey_responses add column if not exists rating integer;
update survey_responses set rating = 5 where rating is null;
alter table survey_responses alter column rating set not null;
do $$ begin
  alter table survey_responses add constraint survey_responses_rating_check check (rating between 1 and 5);
exception when duplicate_object then null;
end $$;
do $$ begin
  if exists (select 1 from information_schema.columns
             where table_name='survey_responses' and column_name='has_issue') then
    alter table survey_responses alter column has_issue drop not null;
  end if;
  if exists (select 1 from information_schema.columns
             where table_name='survey_responses' and column_name='suggestion') then
    alter table survey_responses alter column suggestion drop not null;
  end if;
end $$;

-- 10) EVENT KEMERDEKAAN (banner + Plus gratis massal) — /owner (Event Panel)
--     dipakai lib/event.ts, /api/owner/event, /api/event/status
-- ----------------------------------------------------------------------------
-- Cuma 1 baris (id = 1) yang dipakai — nyimpen status event yang lagi
-- berjalan (atau terakhir berjalan). active/started_at/ends_at = kondisi
-- SEKARANG. scheduled_start_at/scheduled_end_at = jadwal MASA DEPAN yang
-- belum kejadian (buat auto-aktif tanpa owner perlu pencet tombol pas
-- momennya) — begitu jadwal itu kepakai (auto-aktif kejalan), otomatis
-- dikosongin lagi. Dicek "lazy" (kayak reset komentar announcement), bukan
-- cron beneran — jadi dicek ulang tiap ada yang buka banner/status/panel.
create table if not exists event_settings (
  id integer primary key default 1,
  active boolean not null default false,
  hut_number integer not null default 81,
  started_at timestamptz,
  ends_at timestamptz,              -- NULL = gak ada batas waktu, mati manual doang
  scheduled_start_at timestamptz,   -- NULL = gak ada jadwal auto-mulai
  scheduled_end_at timestamptz,     -- dipakai bareng scheduled_start_at buat auto-mulai + auto-selesai
  notified_at timestamptz,          -- kapan notif "event dimulai" terakhir dikirim (anti-dobel)
  updated_by text,
  updated_at timestamptz not null default now(),
  constraint event_settings_single_row check (id = 1)
);
insert into event_settings (id) values (1) on conflict (id) do nothing;

-- 16) MESSAGE (chat user <-> owner/admin) — /messages, /api/messages/*
-- ----------------------------------------------------------------------------
-- User BEBAS milih mau chat ke siapa (owner atau admin manapun), tapi
-- owner/admin CUMA BISA BALAS obrolan yang UDAH DIMULAI user (gak bisa
-- mulai obrolan baru duluan) — ini ditegakkan di kode (lib/messages.ts),
-- bukan di level SQL.
--
-- 1 baris di dm_conversations = 1 "ruang obrolan" antara 1 user & 1
-- owner/admin. Kalau owner/admin klik "Akhiri Obrolan", baris ini
-- (+ semua pesannya, lewat on delete cascade) DIHAPUS PERMANEN dari
-- Supabase — bukan cuma ditandai "selesai". Kalau user chat lagi ke
-- owner/admin yang sama abis itu, otomatis kebentuk conversation BARU
-- (obrolan lama beneran bersih, bukan nyambung ke history sebelumnya).
create table if not exists dm_conversations (
  id uuid primary key default gen_random_uuid(),
  user_login text not null,   -- yang MULAI obrolan (selalu user biasa, bukan admin/owner)
  admin_login text not null,  -- owner/admin yang dituju
  created_at timestamptz not null default now(),
  last_message_at timestamptz not null default now(),
  unique (user_login, admin_login)
);
create index if not exists dm_conversations_user_idx on dm_conversations (user_login, last_message_at desc);
create index if not exists dm_conversations_admin_idx on dm_conversations (admin_login, last_message_at desc);

create table if not exists dm_messages (
  id uuid primary key default gen_random_uuid(),
  conversation_id uuid not null references dm_conversations(id) on delete cascade,
  sender_login text not null,
  content text not null,
  created_at timestamptz not null default now()
);
create index if not exists dm_messages_conversation_idx on dm_messages (conversation_id, created_at);

-- ============================================================================
-- 17) LOGIN EMAIL/USERNAME + VERIFIKASI EMAIL WAJIB (fitur baru)
--     Dipakai lib/localAccounts.ts, lib/emailVerification.ts, lib/auth.ts,
--     app/api/auth/register, app/api/auth/verify-email, app/register,
--     app/verify-email.
--
--     KONSEP: sekarang ada 2 cara login yang BERDAMPINGAN:
--       1. GitHub (OAuth, seperti sebelumnya)
--       2. Username/Password lokal (akun baru, disimpan di local_accounts)
--     Akun lokal BISA ditautkan (link) ke akun GitHub biar dapat akses fitur
--     yang butuh GitHub (editor/repo/dll). Kedua jenis akun WAJIB verifikasi
--     email sebelum bisa pakai KRYNOS sama sekali.
-- ============================================================================

create table if not exists local_accounts (
  id uuid primary key default gen_random_uuid(),
  username text not null,
  username_lower text not null unique,
  password_hash text not null,
  email text not null,
  email_lower text not null,
  email_verified boolean not null default false,
  status text not null default 'pending', -- 'pending' (belum verif email) | 'active' | 'banned'
  github_login text, -- diisi kalau akun ini udah ditautkan ke akun GitHub
  created_at timestamptz not null default now()
);
create unique index if not exists local_accounts_email_verified_idx
  on local_accounts (email_lower) where email_verified = true;
create index if not exists local_accounts_github_idx on local_accounts (github_login);

-- Email yang ditautkan ke akun GitHub (akun GitHub gak punya kolom email
-- bawaan di app ini, jadi disimpan terpisah di sini). Akun GitHub LAMA yang
-- belum punya baris di sini akan diblokir aksesnya sampai isi + verifikasi
-- email lewat halaman /verify-email.
create table if not exists github_account_emails (
  github_login text primary key,
  email text not null,
  email_lower text not null,
  email_verified boolean not null default false,
  created_at timestamptz not null default now()
);
create unique index if not exists github_account_emails_verified_idx
  on github_account_emails (email_lower) where email_verified = true;

-- Kode verifikasi (6 digit) + token, dipakai dua-duanya (local_accounts &
-- github_account_emails) lewat target_type/target_id.
create table if not exists email_verifications (
  token text primary key,
  target_type text not null, -- 'local' | 'github'
  target_id text not null,   -- local_accounts.id (uuid as text) ATAU github login
  email text not null,
  code text not null,        -- 6 digit, biar gampang diketik manual juga
  expires_at timestamptz not null,
  created_at timestamptz not null default now()
);
create index if not exists email_verifications_target_idx
  on email_verifications (target_type, target_id);

);
create index if not exists password_resets_account_idx on password_resets (local_account_id);

-- Perangkat yang PERNAH login ke akun ini (fingerprint dari User-Agent) —
-- dipakai buat deteksi "perangkat baru connect ke akun". Bukan pelacakan
-- presisi (User-Agent bisa sama antar device sejenis), tapi cukup buat
-- ngasih tau kalau ada login dari browser/OS yang belum pernah kepakai di
-- akun ini sebelumnya.
create table if not exists known_devices (
  id uuid primary key default gen_random_uuid(),
  identity_key text not null,   -- login GitHub, ATAU "local:<username>" (lihat lib/identity.ts)
  device_fingerprint text not null,
  device_label text,            -- contoh: "Chrome di Android"
  first_seen timestamptz not null default now(),
  last_seen timestamptz not null default now(),
  unique (identity_key, device_fingerprint)
);
create index if not exists known_devices_identity_idx on known_devices (identity_key);

-- Notifikasi keamanan yang muncul di NotificationBell (ganti password,
-- perangkat baru connect, dll) — SENGAJA GAK ADA notifikasi buat "lupa
-- password" (request reset password gak nulis apa-apa ke tabel ini, sesuai
-- permintaan).
create table if not exists security_notifications (
  id uuid primary key default gen_random_uuid(),
  identity_key text not null,   -- sama kayak known_devices.identity_key
  type text not null,           -- 'password_changed' | 'new_device'
  message text not null,
  created_at timestamptz not null default now()
);
create index if not exists security_notifications_identity_idx
  on security_notifications (identity_key, created_at desc);

-- ============================================================================
-- 20) APLIKASI PIHAK KETIGA (Vercel/Netlify) — /third-party-apps,
--     lib/thirdPartyApps.ts, /api/vercel-connect/*, /api/netlify-connect/*
--
--     PENTING: tabel ini TERPISAH TOTAL dari VERCEL_TOKEN/NETLIFY_TOKEN yang
--     udah ada sebelumnya (punya OWNER KRYNOS sendiri, dipakai fitur
--     Deployment lama di lib/vercel.ts & lib/netlify.ts). Tabel ini nyimpen
--     token milik MASING-MASING USER KRYNOS (hasil OAuth "Connect
--     Vercel/Netlify" dari akun mereka sendiri), jadi tiap user cuma bisa
--     ngatur project/site di AKUN MEREKA SENDIRI, gak akan ketuker atau
--     numpuk ke akun siapapun yang lain.
-- ============================================================================
create table if not exists third_party_connections (
  id uuid primary key default gen_random_uuid(),
  identity_key text not null,   -- sama kayak known_devices.identity_key (login GitHub / "local:username")
  provider text not null,       -- 'vercel' | 'netlify'
  access_token text not null,
  refresh_token text,           -- boleh null, gak semua OAuth app dikasih ini
  token_type text,
  scope text,
  -- Metadata akun (buat ditampilin di halaman "Aplikasi Pihak Ketiga" tanpa
  -- perlu nembak API lagi tiap render, sekaligus buat konteks kalau ada
  -- error/limit yang perlu ditunjukin apa adanya ke user):
  provider_account_id text,     -- user_id (Vercel) / account id (Netlify)
  provider_team_id text,        -- team_id (Vercel doang, null kalau personal account)
  account_email text,
  account_name text,
  connected_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  unique (identity_key, provider)
);
create index if not exists third_party_connections_identity_idx
  on third_party_connections (identity_key);

-- Nyimpen "state" sementara pas lagi di tengah alur OAuth (dari klik
-- "Connect" sampai balik dari halaman approve Vercel/Netlify) — dipakai
-- buat proteksi CSRF (state di callback harus cocok sama yang disimpan di
-- sini, baru dianggap valid). Baris otomatis kadaluarsa 10 menit, dibuang
-- pas berhasil dipakai ATAU pas kadaluarsa (dicek manual di route, Supabase
-- gak punya auto-expire bawaan buat tabel biasa).
create table if not exists third_party_oauth_states (
  state text primary key,
  identity_key text not null,
  provider text not null,
  created_at timestamptz not null default now()
);

-- ============================================================================
-- 21) DAILY REWARDS — ikon api di sidebar (lib/dailyRewards.ts,
--     /api/daily-rewards, components/DailyRewardsFlame.tsx)
--
--     Key-nya SAMA PERSIS kayak `login` yang dipakai tabel plus_subscriptions
--     (lib/plus.ts) — biar reward-nya (nambah durasi Plus) bisa langsung
--     manggil fungsi yang sama (grantPlusMinutes), gak perlu tabel/relasi
--     terpisah buat nyambungin ke akun Plus user.
--
--     "Hari" dihitung pake tanggal WIB (Asia/Jakarta), BUKAN UTC atau timezone
--     device masing-masing user — biar konsisten buat semua orang gak peduli
--     mereka di zona waktu mana pas check-in.
-- ============================================================================
create table if not exists daily_checkins (
  login text primary key,
  streak integer not null default 0,
  last_checkin_date text not null, -- format 'YYYY-MM-DD', tanggal WIB
  updated_at timestamptz not null default now()
);

-- ============================================================================
-- 22) PASSKEY — login pakai sidik jari/Face ID/Windows Hello, OPSIONAL per
--     user (lib/passkey.ts, /api/passkey/*). Cuma bisa dipasang ke AKUN
--     LOKAL (username/password dari lib/localAccounts.ts) — bukan akun
--     GitHub OAuth, karena GitHub sendiri udah punya sistem passkey-nya
--     sendiri di luar KRYNOS.
-- ============================================================================
create table if not exists passkey_credentials (
  id text primary key,              -- credential ID (base64url) dari authenticator
  local_account_id text not null,   -- FK ke local_accounts.id
  public_key text not null,         -- public key, disimpen base64
  webauthn_user_id text not null,   -- "user.id" internal WebAuthn (beda dari local_account_id)
  counter bigint not null default 0,
  device_type text not null,        -- 'singleDevice' | 'multiDevice'
  backed_up boolean not null default false,
  transports text,                  -- CSV, mis. "internal,hybrid"
  device_name text,                 -- label biar user gampang kenalin ("iPhone Ridho")
  created_at timestamptz not null default now()
);
create index if not exists passkey_credentials_account_idx
  on passkey_credentials (local_account_id);

-- Challenge sementara pas lagi proses registrasi ATAU login pakai passkey —
-- 1 baris aktif per akun (di-upsert/di-timpa tiap mulai proses baru), buang
-- otomatis (dicek manual expiry-nya di kode) kalau lebih dari 5 menit.
create table if not exists passkey_challenges (
  local_account_id text primary key,
  purpose text not null, -- 'register' | 'authenticate'
  challenge text not null,
  created_at timestamptz not null default now()
);

-- ============================================================================
-- 23) TEMPLATE — sidebar "Template". Owner nambah link repo GitHub publik +
--     judul; user klik "Get" buat auto-star repo sumber lalu copy isinya
--     jadi repo baru punya user (lib/templates.ts, /api/templates/*,
--     /api/owner/templates/*).
--
--     repo_owner & repo_name SENGAJA gak pernah dikirim ke client biasa
--     (cuma dipakai server-side di /api/templates/[id]/get) -- itu yang
--     bikin link repo sumber gak kelihatan sama user.
-- ============================================================================
create table if not exists templates (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  repo_owner text not null,
  repo_name text not null,
  created_by text not null,
  created_at timestamptz not null default now()
);
